# 10. Color Detection (red or blue)
# =====================================================================
import cv2
import matplotlib.pyplot as plt
 
img = cv2.imread("input.jpg")
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
 
color = "red"        # change to "blue" if needed
 
if color == "red":
    mask1 = cv2.inRange(hsv, (0, 120, 70), (10, 255, 255))
    mask2 = cv2.inRange(hsv, (170, 120, 70), (180, 255, 255))
    mask = cv2.bitwise_or(mask1, mask2)
else:
    mask = cv2.inRange(hsv, (100, 150, 50), (130, 255, 255))
 
detected = cv2.bitwise_and(img, img, mask=mask)
 
plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
plt.subplot(1, 3, 2); plt.imshow(mask, cmap="gray"); plt.title("Mask"); plt.axis("off")
plt.subplot(1, 3, 3); plt.imshow(cv2.cvtColor(detected, cv2.COLOR_BGR2RGB)); plt.title("Detected Region"); plt.axis("off")
plt.show()

# 13. Salt-and-Pepper Noise + Median Filter
# =====================================================================
import cv2
import numpy as np
import matplotlib.pyplot as plt
 
gray = cv2.imread("input.jpg", cv2.IMREAD_GRAYSCALE)
noisy = gray.copy()
h, w = gray.shape
num = int(0.05 * h * w / 2)          # 5% noise
 
# salt (white pixels)
ys = np.random.randint(0, h, num)
xs = np.random.randint(0, w, num)
noisy[ys, xs] = 255
 
# pepper (black pixels)
ys = np.random.randint(0, h, num)
xs = np.random.randint(0, w, num)
noisy[ys, xs] = 0
 
cleaned = cv2.medianBlur(noisy, 3)
 
plt.subplot(1, 3, 1); plt.imshow(gray, cmap="gray"); plt.title("Original"); plt.axis("off")
plt.subplot(1, 3, 2); plt.imshow(noisy, cmap="gray"); plt.title("Noisy"); plt.axis("off")
plt.subplot(1, 3, 3); plt.imshow(cleaned, cmap="gray"); plt.title("Median Filter"); plt.axis("off")
plt.show()

# 15. Object Detection Using Contours
# =====================================================================
import cv2
import matplotlib.pyplot as plt
 
img = cv2.imread("input.jpg")
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
blur = cv2.GaussianBlur(gray, (5, 5), 0)
ret, thresh = cv2.threshold(blur, 0, 255, cv2.THRESH_BINARY_INV + cv2.THRESH_OTSU)
 
contours, hierarchy = cv2.findContours(thresh, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
 
output = img.copy()
count = 0
for c in contours:
    if cv2.contourArea(c) > 100:          # ignore tiny noise
        cv2.drawContours(output, [c], -1, (0, 255, 0), 2)
        count += 1
 
print("Number of objects:", count)
 
plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
plt.subplot(1, 3, 2); plt.imshow(thresh, cmap="gray"); plt.title("Threshold"); plt.axis("off")
plt.subplot(1, 3, 3); plt.imshow(cv2.cvtColor(output, cv2.COLOR_BGR2RGB)); plt.title("Objects: " + str(count)); plt.axis("off")
plt.show()

# 16. Morphological Processing (Erosion and Dilation)
# =====================================================================
import cv2
import numpy as np
import matplotlib.pyplot as plt
 
gray = cv2.imread("input.jpg", cv2.IMREAD_GRAYSCALE)
ret, binary = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)
 
kernel = np.ones((3, 3), np.uint8)
eroded = cv2.erode(binary, kernel, iterations=1)
dilated = cv2.dilate(binary, kernel, iterations=1)
opening = cv2.morphologyEx(binary, cv2.MORPH_OPEN, kernel)     # removes small regions
closing = cv2.morphologyEx(binary, cv2.MORPH_CLOSE, kernel)    # fills small gaps
 
plt.subplot(2, 3, 1); plt.imshow(binary, cmap="gray"); plt.title("Binary"); plt.axis("off")
plt.subplot(2, 3, 2); plt.imshow(eroded, cmap="gray"); plt.title("Erosion"); plt.axis("off")
plt.subplot(2, 3, 3); plt.imshow(dilated, cmap="gray"); plt.title("Dilation"); plt.axis("off")
plt.subplot(2, 3, 4); plt.imshow(opening, cmap="gray"); plt.title("Opening"); plt.axis("off")
plt.subplot(2, 3, 5); plt.imshow(closing, cmap="gray"); plt.title("Closing"); plt.axis("off")
plt.show()

# 18. Face Detection (Haar Cascade)
# =====================================================================
import cv2
import matplotlib.pyplot as plt
 
img = cv2.imread("input.jpg")
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
 
face_cascade = cv2.CascadeClassifier(cv2.data.haarcascades + "haarcascade_frontalface_default.xml")
faces = face_cascade.detectMultiScale(gray, scaleFactor=1.1, minNeighbors=5, minSize=(30, 30))
 
output = img.copy()
for (x, y, w, h) in faces:
    cv2.rectangle(output, (x, y), (x + w, y + h), (0, 255, 0), 2)
 
print("Faces detected:", len(faces))
 
plt.subplot(1, 2, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
plt.subplot(1, 2, 2); plt.imshow(cv2.cvtColor(output, cv2.COLOR_BGR2RGB)); plt.title("Faces: " + str(len(faces))); plt.axis("off")
plt.show()

# 19. Image Difference
# =====================================================================
import cv2
import numpy as np
import matplotlib.pyplot as plt
 
img1 = cv2.imread("input.jpg")
img2 = cv2.imread("input2.jpg")
img2 = cv2.resize(img2, (img1.shape[1], img1.shape[0]))      # both images must be same size
 
gray1 = cv2.cvtColor(img1, cv2.COLOR_BGR2GRAY)
gray2 = cv2.cvtColor(img2, cv2.COLOR_BGR2GRAY)
 
diff = cv2.absdiff(gray1, gray2)
ret, mask = cv2.threshold(diff, 30, 255, cv2.THRESH_BINARY)
mask = cv2.dilate(mask, np.ones((3, 3), np.uint8), iterations=2)
 
contours, hierarchy = cv2.findContours(mask, cv2.RETR_EXTERNAL, cv2.CHAIN_APPROX_SIMPLE)
marked = img2.copy()
for c in contours:
    if cv2.contourArea(c) > 50:
        x, y, w, h = cv2.boundingRect(c)
        cv2.rectangle(marked, (x, y), (x + w, y + h), (0, 0, 255), 2)
 
plt.subplot(2, 2, 1); plt.imshow(cv2.cvtColor(img1, cv2.COLOR_BGR2RGB)); plt.title("Image 1"); plt.axis("off")
plt.subplot(2, 2, 2); plt.imshow(cv2.cvtColor(img2, cv2.COLOR_BGR2RGB)); plt.title("Image 2"); plt.axis("off")
plt.subplot(2, 2, 3); plt.imshow(mask, cmap="gray"); plt.title("Difference Mask"); plt.axis("off")
plt.subplot(2, 2, 4); plt.imshow(cv2.cvtColor(marked, cv2.COLOR_BGR2RGB)); plt.title("Differences Marked"); plt.axis("off")
plt.show()

# 20. Image Segmentation (Green Leaf using HSV)
# =====================================================================
import cv2
import numpy as np
import matplotlib.pyplot as plt
 
img = cv2.imread("input.jpg")
hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
 
lower_green = np.array([35, 40, 40])
upper_green = np.array([85, 255, 255])
mask = cv2.inRange(hsv, lower_green, upper_green)
 
kernel = np.ones((5, 5), np.uint8)
mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, kernel)
 
leaf = cv2.bitwise_and(img, img, mask=mask)
 
plt.subplot(1, 3, 1); plt.imshow(cv2.cvtColor(img, cv2.COLOR_BGR2RGB)); plt.title("Original"); plt.axis("off")
plt.subplot(1, 3, 2); plt.imshow(mask, cmap="gray"); plt.title("Green Mask"); plt.axis("off")
plt.subplot(1, 3, 3); plt.imshow(cv2.cvtColor(leaf, cv2.COLOR_BGR2RGB)); plt.title("Segmented Leaf"); plt.axis("off")
plt.show()
